🧩 Παράδειγμα Κώδικα (Strategy Pattern)
interface CateringCostStrategy {
    double calculateCost(List<Person> people);
}

class StandardCateringCost implements CateringCostStrategy {
    public double calculateCost(List<Person> people) {
        long vegans = people.stream().filter(Person::isVegan).count();
        long nonVegans = people.size() - vegans;
        return nonVegans * 3 + vegans * 3.5;
    }
}

class DiscountCateringCost implements CateringCostStrategy {
    public double calculateCost(List<Person> people) {
        long vegans = people.stream().filter(Person::isVegan).count();
        long nonVegans = people.size() - vegans;
        return nonVegans * 2 + vegans * 2.5;
    }
}

class CateringService {
    private CateringCostStrategy strategy;

    public CateringService(CateringCostStrategy strategy) {
        this.strategy = strategy;
    }

    public double calculateFoodCost(List<Person> people) {
        return strategy.calculateCost(people);
    }
}

import java.util.ArrayList;
import java.util.List;

// Person class (υπόθεση)
class Person {
    private String name;
    private boolean vegan;

    public Person(String name, boolean vegan) {
        this.name = name;
        this.vegan = vegan;
    }

    public boolean isVegan() {
        return vegan;
    }
}

// Strategy interface για κόστος catering
interface CateringCostStrategy {
    double calculateCost(List<Person> people);
}

// Standard κοστολόγηση (<= 500 άτομα)
class StandardCateringCost implements CateringCostStrategy {
    public double calculateCost(List<Person> people) {
        long vegans = people.stream().filter(Person::isVegan).count();
        long nonVegans = people.size() - vegans;
        return nonVegans * 3.0 + vegans * 3.5;
    }
}

// Discount κοστολόγηση (> 500 άτομα)
class DiscountCateringCost implements CateringCostStrategy {
    public double calculateCost(List<Person> people) {
        long vegans = people.stream().filter(Person::isVegan).count();
        long nonVegans = people.size() - vegans;
        return nonVegans * 2.0 + vegans * 2.5;
    }
}

// Service που χρησιμοποιεί την κατάλληλη στρατηγική catering
class CateringService {
    private CateringCostStrategy strategy;

    public CateringService(CateringCostStrategy strategy) {
        this.strategy = strategy;
    }

    public void setStrategy(CateringCostStrategy strategy) {
        this.strategy = strategy;
    }

    public double calculateFoodCost(List<Person> people) {
        return strategy.calculateCost(people);
    }
}

// Service για υπολογισμό κόστους φωτογράφου
class PhotographyService {
    public double calculatePhotoCost(List<Person> people) {
        int count = people.size();
        if (count > 600) {
            return count * 0.75;
        } else {
            return count * 1.0;
        }
    }
}

// Κύρια κλάση Wedding που διαχειρίζεται μόνο venue και people
class Wedding {
    private String venue;
    private List<Person> people;
    private CateringService cateringService;
    private PhotographyService photographyService;

    public Wedding(String venue) {
        this.venue = venue;
        this.people = new ArrayList<>();
        this.cateringService = null; // θα οριστεί ανάλογα
        this.photographyService = new PhotographyService();
    }

    public void addPerson(Person p) {
        this.people.add(p);
        updateCateringStrategy();
    }

    private void updateCateringStrategy() {
        if (people.size() > 500) {
            this.cateringService = new CateringService(new DiscountCateringCost());
        } else {
            this.cateringService = new CateringService(new StandardCateringCost());
        }
    }

    public double calculateFoodCost() {
        if (cateringService == null) {
            updateCateringStrategy();
        }
        return cateringService.calculateFoodCost(people);
    }

    public double calculatePhotoCost() {
        return photographyService.calculatePhotoCost(people);
    }

    public String getVenue() {
        return venue;
    }

    public List<Person> getPeople() {
        return people;
    }
}
