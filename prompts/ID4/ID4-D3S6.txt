import java.util.ArrayList;
import java.util.List;

// Abstract class defining the skeleton of a state machine
abstract class StateMachine {
    public void run() {
        start();
        while (!isFinished()) {
            execute();
        }
        finish();
    }

    protected abstract void start();
    protected abstract void execute();
    protected abstract boolean isFinished();
    protected abstract void finish();
}

// Machine that finds numbers divisible by 4
class DivisibleNumbersMachine extends StateMachine {
    private int counter = 0;
    private final int threshold;

    public DivisibleNumbersMachine(int threshold) {
        this.threshold = threshold;
    }

    @Override
    protected void start() {
        System.out.println("Starting to find numbers divisible by 4 until " + threshold);
    }

    @Override
    protected void execute() {
        if (counter % 4 == 0) {
            System.out.println(counter + " is divisible by 4");
        }
        counter++;
    }

    @Override
    protected boolean isFinished() {
        return counter >= threshold;
    }

    @Override
    protected void finish() {
        System.out.println("Automaton finished");
    }
}

// Machine that uses a stack and computes the sum
class PushDownAutomaton extends StateMachine {
    private int counter = 0;
    private final int threshold;
    private final List<Integer> stack = new ArrayList<>();

    public PushDownAutomaton(int threshold) {
        this.threshold = threshold;

    protected void execute() {

    protected boolean isFinished() {

    }
}

With your original code, you'd need to:

Add a new if (machineType.equals("PrimeNumbersMachine")) check in start(), execute(), isFinished(), and finish()

This violates the Open/Closed Principle (you have to modify existing working code)

Increases coupling and complexity over time

✅ Why the Refactored Design Makes Extension Easy
Using the abstract base class approach (Strategy Pattern via inheritance), adding a new machine involves:

Creating a new subclass of StateMachine

Overriding the 4 required methods: start(), execute(), isFinished(), finish()

📌 You don’t touch the existing DivisibleNumbersMachine, PushDownAutomaton, or the run() logic.

🆕 Example Extension — PrimeNumbersMachine
class PrimeNumbersMachine extends StateMachine {
    private int counter = 2;
    private final int threshold;

    public PrimeNumbersMachine(int threshold) {
        this.threshold = threshold;
    }

    @Override
    protected void start() {
        System.out.println("Finding prime numbers up to " + threshold);
    }

    @Override
    protected void execute() {
        if (isPrime(counter)) {
            System.out.println(counter + " is prime");
        }
        counter++;
    }

    @Override
    protected boolean isFinished() {
        return counter > threshold;
    }

    @Override
    protected void finish() {
        System.out.println("Prime number machine finished.");
    }

    private boolean isPrime(int num) {
        if (num < 2) return false;
        for (int i = 2; i <= Math.sqrt(num); i++) {
            if (num % i == 0) return false;
        }
        return true;
    }
}
🔁 In Main.java:
StateMachine primeMachine = new PrimeNumbersMachine(30);
primeMachine.run();
