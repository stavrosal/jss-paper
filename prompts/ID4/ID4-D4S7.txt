// Store ως Composite
class Store {
    private List<Product> products = new ArrayList<>();

    // Προσθήκη οποιουδήποτε προϊόντος
    public void addProduct(Product product) {
        products.add(product);
    }

    // Σύνολο όλων των προϊόντων
    public int getTotalItems() {
        return products.size();
    }

    // Εκτύπωση ονομάτων προϊόντων

class Store extends Product {
    private String name;
    private List<Product> products = new ArrayList<>();

    public Store(String name) {
        this.name = name;
    }

    public void addProduct(Product product) {
        products.add(product);
    }

    public int getTotalItems() {
        int total = 0;
        for (Product p : products) {
            if (p instanceof Store) {
                total += ((Store) p).getTotalItems();
            } else {
                total++;
            }
        }
        return total;
    }

    public void printItemNames() {
        for (Product p : products) {
            if (p instanceof Store) {
                ((Store) p).printItemNames();
            } else {
                System.out.println(p.getName());
            }
        }
    }

    @Override
    public String getName() {
        return name;
    }
}
