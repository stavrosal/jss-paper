import java.util.ArrayList;
import java.util.List;

public class Wedding {
    private String venue;
    private List<Person> people;

    // Constructor
    public Wedding(String venue) {
        this.venue = venue;
        this.people = new ArrayList<>();
    }

    // Add a person to the wedding
    public void addPerson(Person p) {
        this.people.add(p);
    }

    // Calculate total food cost
    public double calculateFoodCost() {
        int vegans = 0;
        for (Person p : people) {
            if (p.isVegan()) {
                vegans++;
            }
        }
        int nonVegans = people.size() - vegans;
        double total;

        if (people.size() <= 500) {
            total = nonVegans * 3.0 + vegans * 3.5;
        } else {
            total = nonVegans * 2.0 + vegans * 2.5;
        }

        return total;
    }

    // Calculate photographer cost
    public double calcPhotosCost() {
        double cost;

        if (people.size() <= 600) {
            cost = people.size() * 1.0;
        } else {
            cost = people.size() * 0.75;
        }

        return cost;
    }
}
