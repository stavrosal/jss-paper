Why it's a perfect fit
You have two "families" of products:

●	Enforces family consistency

    Pants createPants();

import java.util.ArrayList;
import java.util.List;

public class Shop {
    private final ClothesFactory factory;
    private final List<Shirt> shirts = new ArrayList<>();
    private final List<Pants> pants = new ArrayList<>();

    public Shop(ClothesFactory factory) {
        this.factory = factory;
    }

    public void addShirt() {
        shirts.add(factory.createShirt());
    }

    public void addPants() {
        pants.add(factory.createPants());
    }

    public void printInventory() {
        shirts.forEach(Shirt::printInfo);
        pants.forEach(Pants::printInfo);
    }
}
